cmake_minimum_required(VERSION 3.24)
project(CyMCPChatUIConsumer LANGUAGES CXX)
set(CYUI_BUILD_DEMO OFF)
set(CYUI_BUILD_AUDIO OFF)
set(CYUI_BUILD_GALLERY OFF)
set(CYUI_BUILD_TESTS OFF)
set(CYUI_MODULES DocumentViews)
add_subdirectory("${CYMCP_CYUI_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cyui" EXCLUDE_FROM_ALL)
find_package(CyMCPSDK 0.5 CONFIG REQUIRED COMPONENTS ChatUI)
add_executable(chat_ui_consumer main.cpp)
target_link_libraries(chat_ui_consumer PRIVATE CyMCP::ChatUI)
enable_testing()
add_test(NAME installed_chat_ui COMMAND chat_ui_consumer)
